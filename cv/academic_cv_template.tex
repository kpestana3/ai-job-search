%% Academic CV - [YOUR_NAME]
%% Full, untrimmed academic CV: complete publication list, all conference
%% presentations, teaching/mentoring experience, and grants. Unlike
%% cv/main_example.tex (the 2-page industry resume), this document has NO
%% page limit -- content is reordered/reweighted per posting to surface
%% relevant keywords, but never cut for length.
%%
%% SETUP: Run /setup to populate this with your actual information.
%%
%% Standard section order (see 05-cv-templates.md "Academic CV Section Order"):
%% Professional Statement, Research Skills, Work Experience,
%% International Training, Awards and Funding, Teaching and Mentoring,
%% Education, Publications and Posters, then Service and Leadership,
%% Professional Memberships, References.
%%
%% Every \section is preceded by \needspace{...} sized for the heading plus
%% at least its first entry, so a heading can never be orphaned at the
%% bottom of a page with its content pushed entirely to the next page.
%%
%% Compile with: lualatex academic_cv_template.tex
%% (pdflatex often fails on modern MiKTeX with fontawesome5 font-expansion errors.)

\documentclass[11pt,letterpaper,sans]{moderncv}
\moderncvstyle{banking}
\moderncvcolor{blue}
% Force the marvosym icon set instead of banking's lualatex/xelatex default
% (fontawesome5, via moderncviconsawesome.sty). fontawesome5 icons here
% render as literal fallback text ("MOBILE-ANDROID-ALT") in the compiled
% PDF under this MiKTeX+lualatex setup -- the same glyph-name-leak root
% cause already fixed in the ats-resume template's header, but worse here
% since it shows in the rendered page itself, not just pdftotext output.
\moderncvicons{marvosym}
% moderncv's default header separator between phone/email/extrainfo is
% \textbullet forced into \rmfamily (see moderncvheadiii.sty's
% \makeheaddetailssymbol) -- under this MiKTeX+lualatex setup that glyph
% extracts as a bare U+FFFD replacement character via pdftotext, even
% though the icon fix above already resolved the visible FontAwesome
% fallback-text bug. Override to a plain pipe, which extracts cleanly and
% matches the separator style already used in the ats-resume template's
% contact line.
\renewcommand*{\makeheaddetailssymbol}{\hspace{1em}{\rmfamily |}\hspace{1em}}

% Force both first and last name AND section headings to render in moderncv
% blue (color1). Default banking on lualatex+MiKTeX leaves these black, which
% looks inconsistent with the rest of the blue accent scheme.
\renewcommand*{\firstnamestyle}[1]{{\fontsize{34}{36}\bfseries\upshape\color{color1}#1}}
\renewcommand*{\lastnamestyle}[1]{{\fontsize{34}{36}\bfseries\upshape\color{color1}#1}}
\renewcommand*{\sectionstyle}[1]{{\sectionfont\color{color1}#1}}

\usepackage[utf8]{inputenc}
\usepackage{hyperref}
\usepackage{needspace}
\hypersetup{
    colorlinks=true,
    linkcolor=blue,
    filecolor=magenta,
    urlcolor=blue,
    pdftitle={[YOUR_NAME] - Academic CV},
    pdfpagemode=FullScreen,
}
% Explicit, symmetric margins instead of geometry's scale=0.80 -- scale
% mode can leave the bottom margin visibly larger than the top, likely
% because moderncv's own headheight/headsep eats into geometry's top-margin
% share differently than footskip eats into the bottom. Explicit top=bottom
% values remove that ambiguity.
\usepackage[top=1in, bottom=1in, left=0.75in, right=0.75in]{geometry}
\usepackage{import}
